const [users, setUsers] = useState<RegisteredUser[]>([]); const
[userSearch, setUserSearch] = useState(““);

localStorage.getItem(“ambapari_users”); setUsers(raw ? JSON.parse(raw) :
[]); } catch { setUsers([]); } };

load(); window.addEventListener(“storage”, load);

return () => window.removeEventListener(“storage”, load); }, []);
